+def soften_subject(pixels: torch.Tensor, hole: torch.Tensor, blur: int, level: float | None = None,
+                   gain: float = 1.0, colour: bool = False) -> torch.Tensor:
+    """[F, H, W, 3] with the pixels under `hole` [F, H, W] reduced to their coarse shape: no colour, blurred.
+
+    EXPERIMENT, 2026-10-04 (mrhf): a late start on the schedule carries the
+    original's pose into the render and its hair and clothes with it
+    (`docs/research/masking/2026-10-04_mrhf.md`). This is what the late start
+    is given in place of the original inside the region: luminance only,
+    blurred by a Gaussian of `blur` pixels that averages inside the hole only,
+    so the background does not bleed in. Outside the hole nothing changes.
+
+    With `level`, each frame's blur is moved so that its mean under the hole
+    is `level`, and its contrast about that mean is scaled by `gain`: how
+    light or dark the original's clothes were goes, where things are stays.
+    """
+    sigma = max(float(blur), 0.5)
+    radius = int(3 * sigma)
+    k = torch.exp(-0.5 * (torch.arange(-radius, radius + 1, dtype=torch.float32) / sigma) ** 2)
+    k = (k / k.sum()).to(pixels.device, pixels.dtype)
+
+    def smooth(x: torch.Tensor) -> torch.Tensor:        # [N, 1, H, W]
+        x = F.conv2d(F.pad(x, (radius, radius, 0, 0), mode="replicate"), k.view(1, 1, 1, -1))
+        return F.conv2d(F.pad(x, (0, 0, radius, radius), mode="replicate"), k.view(1, 1, -1, 1))
+
+    out = pixels.clone()
+    for i in range(0, pixels.shape[0], CHUNK):
+        h = (hole[i:i + CHUNK] > 0.5).to(pixels.dtype).unsqueeze(1)
+        if not bool(h.any()):
+            continue
+        if colour:      # EXPERIMENT: the blur keeps its colour, so the original's clothes keep theirs
+            rgb = pixels[i:i + CHUNK, ..., :3].movedim(-1, 1)
+            weight = smooth(h).clamp(min=1e-4)
+            soft = torch.cat([smooth(rgb[:, c:c + 1] * h) / weight for c in range(3)], dim=1)
+            img = pixels[i:i + CHUNK].movedim(-1, 1)
+            out[i:i + CHUNK] = (img * (1.0 - h) + soft * h).movedim(1, -1)
+            continue
+        grey = pixels[i:i + CHUNK, ..., :3].mean(dim=-1).unsqueeze(1)
+        soft = smooth(grey * h) / smooth(h).clamp(min=1e-4)
+        if level is not None:
+            mean = (soft * h).sum(dim=(1, 2, 3), keepdim=True) / h.sum(dim=(1, 2, 3), keepdim=True).clamp(min=1.0)
+            soft = (float(level) + float(gain) * (soft - mean)).clamp(0.0, 1.0)
+        img = pixels[i:i + CHUNK].movedim(-1, 1)
+        out[i:i + CHUNK] = (img * (1.0 - h) + soft.expand(-1, img.shape[1], -1, -1) * h).movedim(1, -1)
+    return out
+
